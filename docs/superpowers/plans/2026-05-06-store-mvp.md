# Store MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implementar o MVP público da Vello Store: landing page com produtos, simulador, formulário de solicitação em 2 passos e tela de confirmação criando um registro `proposals` com `status = 'simulation'`.

**Architecture:** Store completamente pública — sem autenticação. Todas as queries usam `createServiceClient()` (service key server-side). O `tenant_id` vem exclusivamente de `headers().get('x-tenant-id')`, injetado pelo `proxy.ts`. A transição entre os passos do formulário usa query params na URL; o passo 2 transporta os dados do passo 1 como hidden inputs no FormData do `createProposal`.

**Tech Stack:** Next.js 16 App Router, TypeScript, Supabase (service client), Zod, shadcn/ui (Button, Input, Select, Slider, Label, Badge), `@react-input/mask`, ViaCEP API (via Route Handler), Tailwind CSS com tokens semânticos.

**Spec:** `docs/superpowers/specs/2026-05-06-store-design.md`

```sql
  FOR EACH ROW EXECUTE FUNCTION set_updated_at();

CREATE INDEX idx_proposals_tenant_created ON proposals(tenant_id, created_at DESC);

ALTER TABLE proposals ENABLE ROW LEVEL SECURITY;

CREATE POLICY "backoffice_full_access" ON proposals
  FOR ALL USING (

  FOR ALL USING (
```

```typescript
// apps/store/src/lib/utils/cpf.ts

export function validateCPF(raw: string): boolean {
  const digits = raw.replace(/\D/g, '')
  if (digits.length !== 11) return false
  if (/^(\d)\1{10}$/.test(digits)) return false

  let sum = 0
  for (let i = 0; i < 9; i++) sum += Number(digits[i]) * (10 - i)
  let rem = (sum * 10) % 11
  if (rem === 10 || rem === 11) rem = 0
  if (rem !== Number(digits[9])) return false

  sum = 0
  for (let i = 0; i < 10; i++) sum += Number(digits[i]) * (11 - i)
  rem = (sum * 10) % 11
  if (rem === 10 || rem === 11) rem = 0
  return rem === Number(digits[10])
}
```

```typescript
// apps/store/src/lib/utils/simulator.ts

/** Gera até 5 opções de prazo em intervalos de 12 meses. */
export function generateTermOptions(min: number, max: number): number[] {
  if (min === max) return [min]

  const byStep: number[] = []
  for (let t = min; t <= max; t += 12) byStep.push(t)
  if (byStep[byStep.length - 1] !== max) byStep.push(max)

  if (byStep.length <= 5) return byStep

  // Distribui 5 pontos igualmente
  const result = [min]
  const interval = (max - min) / 4
  for (let i = 1; i < 4; i++) {
    result.push(Math.round((min + interval * i) / 12) * 12)
  }
  result.push(max)
  return result
}

export function formatCurrency(value: number): string {
  return new Intl.NumberFormat('pt-BR', {
    style: 'currency',
    currency: 'BRL',
  }).format(value)
}

export function formatRate(rate: number): string {
  return rate.toFixed(2).replace('.', ',')
}
```

```typescript
export async function listActiveProducts(tenantId: string) {
  const supabase = createServiceClient()
  const { data } = await supabase
    .from('tenant_products')
    .select(`
      id,
      cover_url,
      products (
        id,
        name,
        type,
        description,
        min_amount,
        max_amount,
        min_term_months,
        max_term_months,
        min_rate_monthly,
        max_rate_monthly
      )
    `)
    .eq('tenant_id', tenantId)
    .eq('is_active', true)
    .order('created_at')
  return data ?? []
}

export async function getTenantProduct(tenantProductId: string, tenantId: string) {
  const supabase = createServiceClient()
  const { data } = await supabase
    .from('tenant_products')
    .select(`
      id,
      cover_url,
      products (
        id,
        name,
        type,
        description,
        min_amount,
        max_amount,
        min_term_months,
        max_term_months,
        min_rate_monthly,
        max_rate_monthly
      )
    `)
    .eq('id', tenantProductId)
    .eq('tenant_id', tenantId)
    .eq('is_active', true)
    .single()
  return data
}

export async function getProposalConfirmation(proposalId: string, tenantId: string) {
  const supabase = createServiceClient()
  const { data } = await supabase
    .from('proposals')
    .select(`
      id,
      client_name,
      desired_amount,
      desired_term_months,
      submitted_at,
      products ( name )
    `)
    .eq('id', proposalId)
    .eq('tenant_id', tenantId)
    .single()
  return data
}
```

```typescript
import { NextRequest, NextResponse } from 'next/server'

export async function GET(
  _req: NextRequest,
  { params }: { params: Promise<{ cep: string }> },
) {
  const { cep } = await params
  const digits = cep.replace(/\D/g, '')

  if (digits.length !== 8) {
    return NextResponse.json({ error: 'CEP inválido' }, { status: 400 })
  }

  try {
    const res = await fetch(`https://viacep.com.br/ws/${digits}/json/`, {
      next: { revalidate: 86400 }, // cache 24h
    })

    if (!res.ok) {
      return NextResponse.json({ error: 'CEP não encontrado' }, { status: 404 })
    }

    const data = (await res.json()) as {
      erro?: boolean
      logradouro: string
      bairro: string
      localidade: string
      uf: string
    }

    if (data.erro) {
      return NextResponse.json({ error: 'CEP não encontrado' }, { status: 404 })
    }

    return NextResponse.json({
      street: data.logradouro,
      neighborhood: data.bairro,
      city: data.localidade,
      state: data.uf,
    })
  } catch {
    return NextResponse.json({ error: 'Erro ao consultar CEP' }, { status: 500 })
  }
}
```

```tsx
// apps/store/src/app/(store)/solicitar/[id]/page.tsx
import type { Metadata } from 'next'
import { notFound } from 'next/navigation'
import { headers } from 'next/headers'
import Image from 'next/image'
import { getTenantBranding } from '@/lib/branding'
import { getTenantProduct } from '../../queries'
import { createProposal } from './actions'
import { StepIndicator } from './_components/step-indicator'
import { StepOneForm } from './_components/step-one-form'
import { StepTwoForm } from './_components/step-two-form'

export const metadata: Metadata = { title: 'Solicitar crédito' }

type Props = {
  params: Promise<{ id: string }>
  searchParams: Promise<{
    step?: string
    purpose?: string
    amount?: string
    term?: string
    name?: string
    email?: string
  }>
}

export default async function RequestPage({ params, searchParams }: Props) {
  const { id } = await params
  const sp = await searchParams

  const headerStore = await headers()
  const tenantId = headerStore.get('x-tenant-id') ?? ''

  const [tp, branding] = await Promise.all([
    getTenantProduct(id, tenantId),
    getTenantBranding(tenantId),
  ])

  if (!tp || !tp.products) notFound()

  const step = sp.step === '2' ? 2 : 1
  const minRate = tp.products.min_rate_monthly ?? 0
  const initialAmount = Number(sp.amount) || tp.products.min_amount || 1000
  const initialTerm = Number(sp.term) || tp.products.min_term_months || 12

  // Ação vinculada ao tenantProductId e productId
  const boundAction = createProposal.bind(null, id, tp.products.id)

  const step1Data = {
    purpose: sp.purpose ?? '',
    amount: sp.amount ?? String(initialAmount),
    term: sp.term ?? String(initialTerm),
    name: sp.name ?? '',
    email: sp.email ?? '',
  }

  return (
    <div className="flex min-h-svh">
      {/* Painel esquerdo — branding */}
      <aside className="relative hidden w-[35%] flex-col justify-between overflow-hidden bg-gradient-to-br from-primary/80 to-primary p-10 lg:flex">
        {branding?.auth_bg_image_url && (
          <Image
            src={branding.auth_bg_image_url}
            alt=""
            fill
            className="object-cover opacity-20"
          />
        )}
        <div className="relative z-10">
          {branding?.logo_url ? (
            <Image
              src={branding.logo_url}
              alt={branding.company_display_name ?? ''}
              width={120}
              height={40}
              className="h-10 w-auto object-contain brightness-0 invert"
            />
          ) : (
            <span className="text-xl font-bold text-primary-foreground">
              {branding?.company_display_name ?? 'Vello'}
            </span>
          )}
        </div>
        <div className="relative z-10 space-y-4">
          <h2 className="text-2xl font-bold text-primary-foreground">
            {tp.products.name}
          </h2>
          <p className="text-primary-foreground/80 text-sm leading-relaxed">
            Preencha seus dados para receber a análise da sua proposta.
          </p>
        </div>
      </aside>

      {/* Painel direito — formulário */}
      <div className="flex flex-1 items-start justify-center px-4 py-12 sm:px-8">
        <div className="w-full max-w-md space-y-8">
          <div>
            <StepIndicator currentStep={step} />
            <h1 className="mt-4 text-xl font-bold text-foreground">
              {step === 1 ? 'Dados da simulação' : 'Dados pessoais'}
            </h1>
            <p className="text-sm text-muted-foreground">
              {step === 1
                ? 'Informe os detalhes do crédito que você precisa.'
                : 'Precisamos de algumas informações para continuar.'}
            </p>
          </div>

          {step === 1 ? (
            <StepOneForm
              tenantProductId={id}
              minRate={minRate}
              initialAmount={initialAmount}
              initialTerm={initialTerm}
            />
          ) : (
            <StepTwoForm
              tenantProductId={id}
              action={boundAction}
              step1Data={step1Data}
            />
          )}
        </div>
      </div>
    </div>
  )
}
```
